    // ==================== KHI BẤM NÚT "NHẬN VÉ e-PASS MỚI (+1 XE)" Ở BƯỚC 2 ====================
    function issueSmartPass() {
      if (!state.docValid || !state.contReady) return;

      state.passSeq++;
      const passId = `PASS-${state.passSeq}`;

      // Tăng số lượng xe trong khung giờ đã chọn
      if (state.slotMode === 'day' && state.dayCount < 60) {
        state.dayCount++;
        const pct = Math.round((state.dayCount / 60) * 100);
        document.getElementById('daySlotCountBadge').innerText = `XANH: ${state.dayCount} / 60 XE`;
        document.getElementById('daySlotProgressBar').style.width = `${pct}%`;
        document.getElementById('dayRemainingText').innerText = `Còn ${60 - state.dayCount} chỗ trống`;
      } else if (state.slotMode === 'night' && state.nightCount < 60) {
        state.nightCount++;
        document.getElementById('nightSlotCountBadge').innerText = `MỞ GATE 10 (${state.nightCount}/60)`;
      }

      // Cộng số liệu tổng ở các cổng bên Tab 2 Web Điều độ
      if (state.slotMode === 'night') {
        state.gate10Total++;
        document.getElementById('opGate10Count').innerText = `${state.gate10Total} xe Cont`;
      } else if (state.truckType === 3) {
        state.gate2Total++;
        document.getElementById('opGate2Count').innerText = `${state.gate2Total} xe (33%)`;
      } else {
        state.gate0Total++;
        document.getElementById('opGate0Count').innerText = `${state.gate0Total} xe (58%)`;
      }

      syncAllScreensAndDashboard();

      // Lưu thêm 1 dòng lịch sử vào bảng Real-time Log bên Tab 2
      const logContainer = document.getElementById('liveBookingLog');
      const historyRow = document.createElement('div');
      historyRow.className = "flex justify-between items-center bg-slate-800/90 border border-slate-700 px-3 py-1.5 rounded-lg";
      historyRow.innerHTML = `
        <span class="font-bold text-slate-200">${passId} | Đã xác nhận: ${state.plate} (${TYPE_NAME_SHORT[state.truckType]})</span>
        <span class="text-blue-400 font-bold">${document.getElementById('activeSyncRight').innerText}</span>`;
      logContainer.appendChild(historyRow);

      // Báo thành công trên Màn hình 2
      const alertBox = document.getElementById('step2AlertBox');
      alertBox.className = "block p-2 rounded-xl text-[10px] font-extrabold text-center bg-emerald-100 text-emerald-900 border border-emerald-500";
      alertBox.innerText = `✓ Đã cấp mã ${passId} cho xe ${state.plate} & đồng bộ sang Tab 2 Web Điều độ!`;

      goToStep(3);
    }
